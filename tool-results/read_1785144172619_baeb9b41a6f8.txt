    57→        original_msg: str,
    58→        guild: discord.Guild,
    59→        channel: discord.TextChannel,
    60→        author: discord.Member,
    61→        referenced_msg: Optional[discord.Message] = None
    62→    ) -> Optional[Tuple[str, bool, Optional[discord.Embed]]]:

    80→            return await self._cmd_show_user_info(author)

    82→        if self._matches_patterns(msg_lower, ["roles", "konse roles", "what roles"]):
    83→            target = target_user or author

   114→            if not self._check_mod(author, guild):
   115→                return ("❌ Sirf mods ban kar sakte hain!", False, None)

   117→            if target_user:

   190→            return await self._cmd_show_help(author)

   194→    def _matches_patterns(self, text: str, patterns: List[str]) -> bool:
   195→        """Check if text matches any of the given patterns"""
   196→        for pattern in patterns:
   197→            if pattern.lower() in text:
   198→                return True
   199→        return False

   201→    async def _extract_target_user(
   202→        self, 
   203→        message: str, 
   204→        channel: discord.TextChannel, 
   205→        guild: discord.Guild,
   206→        referenced_msg: Optional[discord.Message] = None
   207→    ) -> Optional[discord.Member]:
   208→        """Extract target user from mentions, username, or referenced message - IMPROVED!"""
   209→        
   210→        # First check for mentions in message (most reliable)
   211→        mention_pattern = r'<@!?(\d{17,19})>'
   212→        match = re.search(mention_pattern, message)
   213→        if match:
   214→            user_id = int(match.group(1))
   215→            member = channel.guild.get_member(user_id)
   216→            if member:
   217→                return member
   218→        
   219→        # Then check referenced message (reply)
   220→        if referenced_msg:
   221→            return channel.guild.get_member(referenced_msg.author.id)
   222→        
   223→        # NEW: Try to find user by NAME/USERNAME from the message
   224→        # Remove common words and try to match a username
   225→        cleaned_msg = re.sub(r'<@!?\d+>', '', message)  # Remove mentions
   226→        cleaned_msg = re.sub(r'(kick|ban|timeout|mute|isko|iska|usko|use|inhe|inhone|user|ko|ka|ke|liye|karo|kar|do|to|me|se|hai|please|hey|ophelia|@everyone|@here)', '', cleaned_msg, flags=re.IGNORECASE)
   227→        cleaned_msg = cleaned_msg.strip()
   228→        
   229→        if cleaned_msg and len(cleaned_msg) > 2:
   230→            # Search for member by display name or username
   231→            for member in guild.members:
   232→                if cleaned_msg.lower() in member.display_name.lower() or cleaned_msg.lower() in member.name.lower():
   233→                    logger.info(f"🔍 Found user '{cleaned_msg}' → {member} ({member.id})")
   234→                    return member
   235→        
   236→        return None

   238→    def _extract_duration(self, message: str) -> int:
   239→        """Extract timeout duration from message in minutes"""
   240→        patterns = [
   241→            r'(\d+)\s*min',
   242→            r'(\d+)\s*minute',
   243→            r'(\d+)\s*hour',
   244→            r'(\d+)\s*hr',
   245→            r'(\d+)\s*day',
   246→        ]
   247→        
   248→        for pattern in patterns:
   249→            match = re.search(pattern, message.lower())
   250→            if match:
   251→                num = int(match.group(1))
   252→                if 'hour' in pattern or 'hr' in pattern:
   253→                    return num * 60
   254→                elif 'day' in pattern:
   255→                    return num * 1440
   256→                return num
   257→        
   258→        return 10  # Default 10 minutes

   260→    def _extract_reason(self, message: str) -> Optional[str]:
   261→        """Extract reason from message - returns None if no clear reason"""
   262→        reason = message
   263→        for prefix in ['timeout', 'kick', 'ban', 'mute', 'do', 'kar', 'karo']:
   264→            reason = re.sub(prefix, '', reason, flags=re.IGNORECASE)
   265→        
   266→        # Remove mentions and numbers (duration)
   267→        reason = re.sub(r'<@!?\d+>', '', reason)
   268→        reason = re.sub(r'\d+\s*(min|hour|hr|day|minute)?', '', reason, flags=re.IGNORECASE)
   269→        
   270→        reason = reason.strip()
   271→        
   272→        # Only return reason if it looks like actual content (not just random words)
   273→        if len(reason) < 3 or reason.lower() in ['isco', 'iska', 'usko', 'use', 'inhe', 'ko', 'se']:
   274→            return None  # No valid reason found
   275→        
   276→        return reason

   278→    def _extract_activity(self, message: str) -> Dict[str, str]:
   279→        """Extract activity type and name from message"""
   280→        activity_type = "playing"  # Default
   281→        
   282→        if any(word in message.lower() for word in ['listening']):
   283→            activity_type = "listening"
   284→        elif any(word in message.lower() for word in ['watching']):
   285→            activity_type = "watching"
   286→        elif any(word in message.lower() for word in ['streaming']):
   287→            activity_type = "streaming"
   288→        
   289→        name_match = re.search(r'(?:playing|listening||watching|streaming|status)\s+(.+)', message, re.IGNORECASE)
   290→        name = name_match.group(1).strip() if name_match else "with Ophelia AI"
   291→        
   292→        return {"type": activity_type, "name": name}

   294→    def _extract_nickname(self, message: str) -> Optional[str]:
   295→        """Extract new nickname from message"""
   296→        quote_match = re.search(r'"([^"]+)"', message)
   297→        if quote_match:
   298→            return quote_match.group(1)
   299→        
   300→        patterns = [
   301→            r'(?:nickname|name|naam|bolte)\s+(?:to|as|)?\s*(.+)',
   302→            r'(?:call me|bulao|bolo)\s*(.+)',
   303→        ]
   304→        for pattern in patterns:
   305→            match = re.search(pattern, message, re.IGNORECASE)
   306→            if match:
   307→                return match.group(1).strip()[:32]
   308→        
   309→        return None

   311→    def _extract_channel_info(self, message: str) -> Dict[str, str]:
   312→        """Extract channel creation info"""
   313→        name_match = re.search(r'(?:channel)?\s*["\']?(\w[\w\s-]*)["\']?', message, re.IGNORECASE)
   314→        name = name_match.group(1).strip().lower().replace(" ", "-") if name_match else "general"
   315→        
   316→        ch_type = "text"
   317→        if 'voice' in message.lower():
   318→            ch_type = "voice"
   319→        elif 'stage' in message.lower():
   320→            ch_type = "stage"
   321→        
   322→        return {"name": name, "type": ch_type}

   324→    def _extract_role_info(self, message: str) -> Dict[str, str]:
   325→        """Extract role creation info"""
   326→        name_match = re.search(r'(?:role)?\s*["\']?(\w[\w\s]*)["\']?', message, re.IGNORECASE)
   327→        name = name_match.group(1).strip() if name_match else "New Role"
   328→        
   329→        colors = {
   330→            'red': discord.Color.red(),
   331→            'blue': discord.Color.blue(),
   332→            'green': discord.Color.green(),
   333→            'yellow': discord.Color.gold(),
   334→            'purple': discord.Color.purple(),
   335→            'pink': discord.Color.pink(),
   336→            'orange': discord.Color.orange(),
   337→            'white': discord.Color.white(),
   338→            'black': discord.Color.default(),
   339→            'blurple': discord.Color.blurple(),
   340→            'gold': discord.Color.gold(),
   341→        }
   342→        
   343→        color = discord.Color.blue()
   344→        for color_name, color_val in colors.items():
   345→            if color_name in message.lower():
   346→                color = color_val
   347→                break
   348→        
   349→        return {"name": name, "color": color}

   351→    def _extract_send_info(self, message: str, guild: discord.Guild) -> Optional[Dict]:
   352→        """Extract message sending info"""
   353→        channel_match = re.search(r'<#(\d{17,19})>|(?:channel|#)(?:\s+me)?\s*(\w+)', message, re.IGNORECASE)
   354→        
   355→        target_channel = None
   356→        if channel_match:
   357→            if channel_match.group(1):
   358→                target_channel = guild.get_channel(int(channel_match.group(1)))
   359→            else:
   360→                ch_name = channel_match.group(2).lower()
   361→                target_channel = discord.utils.get(guild.text_channels, name=ch_name)
   362→        
   363→        content_match = re.search(r'["\'](.+)["\']|bhejo\s+(.+)|send\s+(.+)|announce\s+(.+)', message, re.IGNORECASE)
   364→        content = content_match.group(1) or content_match.group(2) or content_match.group(3) or content_match.group(4)
   365→        
   366→        if target_channel and content:
   367→            return {"channel": target_channel, "content": content.strip()}
   368→        
   369→        return None

   371→    def _extract_embed_info(self, message: str) -> Optional[Dict]:
   372→        """Extract embed creation info"""
   373→        matches = re.findall(r'"([^"]+)"', message)
   374→        if len(matches) >= 2:
   375→            return {"title": matches[0], "description": matches[1]}
   376→        elif len(matches) == 1:
   377→            return {"title": "Announcement", "description": matches[0]}
   378→        return None

   380→    def _extract_emoji(self, message: str) -> Optional[str]:
   381→        """Extract emoji from message"""
   382→        custom_emoji = re.search(r'<a?:\w+:(\d+)>', message)
   383→        if custom_emoji:
   384→            return custom_emoji.group(0)
   385→        
   386→        emoji_chars = ['😂', '❤️', '🔥', '💀', '✨', '🎉', '👍', '👎', '😢', '😡', '🤔', '💯', '⭐', '🎮', '🎵']
   387→        for emoji in emoji_chars:
   388→            if emoji in message:
   389→                return emoji
   390→        
   391→        return None
   392→    
   393→    def _extract_number(self, message: str) -> Optional[int]:
   394→        """Extract number from message"""
   395→        match = re.search(r'\d+', message)
   396→        return int(match.group()) if match else None
