   421→    
   422→    def _clean_ai_response(self, response: str) -> str:
   423→        """
   424→        Clean AI response - remove raw function calls and weird formatting.
   425→        Prevents leaks like <function=get_channel_info[...]>
   426→        """
   427→        if not response:
   428→            return response
   429→        
   430→        # Remove raw function call syntax: <function=name[params]>
   431→        response = re.sub(r'<function=\w+[\[\{].*?[\]\}]>', '', response, flags=re.DOTALL)
   432→        
   433→        # Remove standalone function calls: function_name(params)
   434→        response = re.sub(r'^\s*\w+\([^)]*\)\s*$', '', response, flags=re.MULTILINE)
   435→        
   436→        # Clean up multiple newlines
   437→        response = re.sub(r'\n{3,}', '\n\n', response)
   438→        
   439→        # Strip whitespace
   440→        response = response.strip()
   441→        
   442→        # Fallback if empty after cleaning
   443→        if not response:
   444→            return "Hmm, kuch gadbad hai! Phir se try karo? 🤔"
   445→        
   446→        return response

   479→    def _detect_mood(self, message: str) -> str:
   480→        """Detect mood from message"""
   481→        msg_lower = message.lower()
   482→        scores = {}
   483→        
   484→        for mood, keywords in self.MOOD_KEYWORDS.items():
   485→            score = sum(1 for kw in keywords if kw in msg_lower)
   486→            scores[mood] = score
   487→        
   488→        best_mood = max(scores, key=scores.get) if any(scores.values()) else "neutral"
   489→        return best_mood

   707→                perms.administrator,
