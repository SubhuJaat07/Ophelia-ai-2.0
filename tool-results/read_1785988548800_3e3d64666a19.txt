    33→    33→    """

    35→    35→    name = "search_messages"

    37→    37→Can find messages by content, author, date range, or keywords.
    38→    38→Perfect for finding past discussions, context, or specific information.
    39→    39→Supports filtering by: content (text search), author_id, channel_id, 
    40→    40→limit (max results), and days_ago (how far back to search)."""
    41→    41→    
    42→    42→    parameters = [
    43→    43→        ToolParameter(
    44→    44→            name="query",
    45→    45→            param_type="string",
    46→    46→            description="Text/keywords to search for in messages",
    47→    47→            required=True

    67→    67→            required=False,

    70→    70→        ToolParameter(
    71→    71→            name="days_ago",
    72→    72→            param_type="integer",
    73→    73→            description="How many days back to search (default: 30, max: 365)",
    74→    74→            required=False,
    75→    75→            default=30
    76→    76→        )
    77→    77→    ]
    78→    78→    
    79→    79→    permission_level = ToolPermissionLevel.EVERYONE
    80→    80→    
    81→    81→    async def execute(self, args: Dict[str, Any], context: Dict[str, Any]) -> ToolResult:
    82→    82→        try:
    83→    83→            query = args["query"].lower()
    84→    84→            channel_id = args.get("channel_id")
    85→    85→            author_id = args.get("author_id")
    86→    86→            limit = min(args.get("limit", 10), 50)
    87→    87→            days_ago = min(args.get("days_ago", 30), 365)
    88→    88→            
    89→    89→            guild = context.get("guild")
    90→    90→            if not guild:
    91→    91→                return ToolResult(success=False, content="❌ No guild/server available in context")
    92→    92→            
    93→    93→            # Calculate date threshold
    94→    94→            since_date = datetime.utcnow() - timedelta(days=days_ago)
    95→    95→            results = []
    96→    96→            
    97→    97→            # Determine channels to search
    98→    98→            if channel_id:
    99→    99→                channel = guild.get_channel(int(channel_id))
   100→   100→                if not channel:
   101→   101→                    return ToolResult(success=False, content=f"❌ Channel {channel_id} not found")
   102→   102→                channels_to_search = [channel] if hasattr(channel, 'history') else []
   103→   103→            else:
   104→   104→                # Search all text channels bot can see
   105→   105→                channels_to_search = [
   106→   106→                    ch for ch in guild.text_channels 
   107→   107→                    if ch.permissions_for(guild.me).read_message_history
   108→   108→                ]
   109→   109→            
   110→   110→            # Search through channels
   111→   111→            for channel in channels_to_search[:20]:  # Limit channels to search
   112→   112→                try:
   113→   113→                    async for message in channel.history(limit=100, after=since_date):
   114→   114→                        # Apply filters
   115→   115→                        if query and query not in message.content.lower():
   116→   116→                            continue
   117→   117→                        if author_id and str(message.author.id) != str(author_id):
   118→   118→                            continue
   119→   119→                        if message.author.bot:  # Skip bot messages
   120→   120→                            continue
   121→   121→                            
   122→   122→                        results.append({
   123→   123→                            "content": message.content[:300],
   124→   124→                            "author": f"{message.author.display_name}",
   125→   125→                            "author_id": str(message.author.id),
   126→   126→                            "channel": f"#{message.channel.name}",
   127→   127→                            "channel_id": str(message.channel.id),
   128→   128→                            "message_id": str(message.id),
   129→   129→                            "created_at": message.created_at.isoformat(),
   130→   130→                            "jump_url": message.jump_url
   131→   131→                        })
   132→   132→                        
   133→   133→                        if len(results) >= limit:
   134→   134→                            break
   135→   135→                    
   136→   136→                    if len(results) >= limit:
   137→   137→                        break
   138→   138→                        
   139→   139→                except Exception as e:
   140→   140→                    logger.debug(f"Cannot search channel {channel}: {e}")
   141→   141→                    continue
   142→   142→            
   143→   143→            if not results:
   144→   144→                return ToolResult(
   145→   145→                    success=True,
   146→   146→                    content=f"😔 No messages found matching '{query}' in the last {days_ago} days.",
   147→   147→                    data={"count": 0, "results": []}
   148→   148→                )
   149→   149→            
   150→   150→            # Format results for AI
   151→   151→            result_lines = [f"🔍 **Found {len(results)} message(s) about '{query}':**\n"]
   152→   152→            for i, msg in enumerate(results[:10], 1):  # Show max 10 in summary
   153→   153→                result_lines.append(
   154→   154→                    f"{i}. **{msg['author']}** in {msg['channel']} ({msg['created_at'][:10]}):\n"
   155→   155→                    f"   > {msg['content'][:200]}\n"
   156→   156→                )
   157→   157→            
   158→   158→            return ToolResult(
   159→   159→                success=True,
   160→   160→                content="\n".join(result_lines),
   161→   161→                data={"count": len(results), "results": results}
   162→   162→            )
   163→   163→            
   164→   164→        except Exception as e:
   165→   165→            logger.error(f"Search failed: {e}")
   166→   166→            return ToolResult(success=False, content=f"❌ Search error: {str(e)[:150]}")

   182→   182→Returns messages with author, timestamp, and content."""
   183→   183→    
   184→   184→    parameters = [
   185→   185→        ToolParameter(
   186→   186→            name="channel_id",
   187→   187→            param_type="string",
   188→   188→            description="ID of the channel to read from",

   195→   195→            required=False,
   196→   196→            default=20
   197→   197→        ),
   198→   198→        ToolParameter(
   199→   199→            name="before_message_id",
   200→   200→            param_type="string",
   201→   201→            description="Get messages before this message ID (for pagination)",
   202→   202→            required=False,
   203→   203→            default=None

   205→   205→    ]
   206→   206→    
   207→   207→    permission_level = ToolPermissionLevel.EVERYONE
   208→   208→    
   209→   209→    async def execute(self, args: Dict[str, Any], context: Dict[str, Any]) -> ToolResult:
   210→   210→        try:
   211→   211→            channel_id = args["channel_id"]
   212→   212→            limit = min(args.get("limit", 20), 100)
   213→   213→            before_id = args.get("before_message_id")
   214→   214→            
   215→   215→            guild = context.get("guild")
   216→   216→            if not guild:
   217→   217→                return ToolResult(success=False, content="❌ No guild available")
   218→   218→            
   219→   219→            channel = guild.get_channel(int(channel_id))
   220→   220→            if not channel:
   221→   221→                return ToolResult(success=False, content=f"❌ Channel {channel_id} not found or not accessible")
   222→   222→            
   223→   223→            if not hasattr(channel, 'history'):
   224→   224→                return ToolResult(success=False, content=f"❌ Cannot read from this channel type")
   225→   225→            
   226→   226→            # Check permissions
   227→   227→            if not channel.permissions_for(guild.me).read_message_history:
   228→   228→                return ToolResult(success=False, content="❌ No permission to read this channel's history")
   229→   229→            
   230→   230→            # Fetch messages
   231→   231→            kwargs = {"limit": limit}
   232→   232→            if before_id:
   233→   233→                kwargs["before"] = discord.Object(id=int(before_id))
   234→   234→            
   235→   235→            messages = []
   236→   236→            async for msg in channel.history(**kwargs):
   237→   237→                messages.append(msg)
   238→   238→            
   239→   239→            if not messages:
   240→   240→                return ToolResult(success=True, content=f"📭 No messages found in #{channel.name}", data=[])
   241→   241→            
   242→   242→            messages.reverse()  # Chronological order
   243→   243→            
   244→   244→            # Format for AI
   245→   245→            lines = [f"📖 **Recent messages from #{channel.name}** ({len(messages)} messages):\n"]
   246→   246→            for msg in messages[-30:]:  # Max 30 in output
   247→   247→                is_bot = "🤖 " if msg.author.bot else ""
   248→   248→                lines.append(
   249→   249→                    f"[{msg.created_at.strftime('%H:%M')}] {is_bot}**{msg.author.display_name}:** {msg.content[:250]}"
   250→   250→                )
   251→   251→            
   252→   252→            return ToolResult(
   253→   253→                success=True,
   254→   254→                content="\n".join(lines),
   255→   255→                data={
   256→   256→                    "count": len(messages),
   257→   257→                    "channel_name": channel.name,
   258→   258→                    "messages": [{
   259→   259→                        "author": m.author.display_name,
   260→   260→                        "content": m.content,
   261→   261→                        "timestamp": m.created_at.isoformat()
   262→   262→                    } for m in messages]
   263→   263→                }
   264→   264→            )
   265→   265→            
   266→   266→        except Exception as e:
   267→   267→            logger.error(f"Read channel failed: {e}")
   268→   268→            return ToolResult(success=False, content=f"❌ Error reading channel: {str(e)[:150]}")

   312→   312→            channel_id = args["channel_id"]
   313→   313→            content = args["message"][:1900]  # Discord limit
   314→   314→            reply_to_id = args.get("reply_to_message_id")

   338→   338→

   340→   340→                success=True,
   341→   341→                content=f"✅ Message sent to #{channel.name}! (ID: {sent_msg.id})",

   344→   344→            
   345→   345→        except discord.Forbidden:
   346→   346→            return ToolResult(success=False, content="❌ Forbidden: Missing permissions to send message")

   348→   348→            logger.error(f"Send message failed: {e}")
   349→   349→            return ToolResult(success=False, content=f"❌ Error sending: {str(e)[:150]}")
