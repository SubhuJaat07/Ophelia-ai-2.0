    29→    async def process_message(
    30→        self,

    32→        guild: discord.Guild,
    33→        channel: discord.TextChannel,

    35→        referenced_message: Optional[discord.Message] = None

    37→        """
    38→        Process user message for natural language commands.
    39→        Returns (response_text, was_command_executed, optional_embed)
    40→        """
    41→        
    42→        msg_lower = message.lower().strip()

    44→        # Check if this looks like a command (action-oriented)

    46→            msg_lower, message, guild, channel, author, referenced_message
    47→        )
    48→        
    49→        if command_result:
    50→            return command_result

    54→    async def _detect_and_execute(
    55→        self,
    56→        msg_lower: str,
    57→        original_msg: str,
    58→        guild: discord.Guild,
    59→        channel: discord.TextChannel,
    60→        author: discord.Member,
    61→        referenced_msg: Optional[discord.Message] = None
    62→    ) -> Optional[Tuple[str, bool, Optional[discord.Embed]]]:
    63→        """Detect and execute natural language commands"""
    64→        
    65→        # Get target user (from mention or referenced message)
    66→        target_user = self._extract_target_user(original_msg, channel, referenced_msg)

    68→        # ===== AVATAR & INFO COMMANDS (Everyone can use) =====

    71→            return await self._cmd_show_avatar(target_user or author, original_msg)
    72→        
    73→        if self._matches_patterns(msg_lower, ["info", "baare me", "about", "details", "profile", "who is"]):
    74→            return await self._cmd_show_user_info(target_user or author)
    75→        
    76→        if self._matches_patterns(msg_lower, ["server info", "server details", "server stats", "kitne members", "member count"]):
    77→            return await self._cmd_show_server_info(guild)
    78→        
    79→        if self._matches_patterns(msg_lower, ["my info", "mera info", "mera profile"]):
    80→            return await self._cmd_show_user_info(author)
    81→        
    82→        if self._matches_patterns(msg_lower, ["roles", "konse roles", "what roles"]):
    83→            target = target_user or author
    84→            return await self._cmd_show_roles(target)
    85→        
    86→        if self._matches_patterns(msg_lower, ["join date", "kab join hua", "when joined"]):
    87→            target = target_user or author
    88→            return await self._cmd_show_join_date(target)
    89→        
    90→        # ===== MODERATION COMMANDS (Owner/Mod Only) =====
    91→        
    92→        if self._matches_patterns(msg_lower, ["timeout", "mute temporarily", "silent mode"]):
    93→            if not self._check_owner(author):
    94→                return ("❌ Sirf owners timeout kar sakte hain!", False, None)
    95→            duration = self._extract_duration(original_msg)
    96→            reason = self._extract_reason(original_msg)

   100→        
   101→        if self._matches_patterns(msg_lower, ["kick", "hatao", "remove", "throw out", "nikalo"]):
   102→            if not self._check_mod(author, guild):
   103→                return ("❌ Sirf mods kick kar sakte hain!", False, None)
   104→            reason = self._extract_reason(original_msg)
   105→            if target_user:
   106→                return await self._cmd_kick(target_user, reason, author, guild)
   107→            return ("❌ Batao kis user ko kick karna hai?", False, None)
   108→        
   109→        if self._matches_patterns(msg_lower, ["ban", "permanent ban", "block", "paka band"]):
   110→            if not self._check_mod(author, guild):
   111→                return ("❌ Sirf mods ban kar sakte hain!", False, None)
   112→            reason = self._extract_reason(original_msg)
   113→            if target_user:
   114→                return await self._cmd_ban(target_user, reason, author, guild)
   115→            return ("❌ Batao kis user ko ban karna hai?", False, None)
   116→        
   117→        if self._matches_patterns(msg_lower, ["mute", "chupao", "silence", "shut up"]):
   118→            if not self._check_mod(author, guild):
   119→                return ("❌ Sirf mods mute kar sakte hain!", False, None)
   120→            if target_user:
   121→                return await self._cmd_mute(target_user, author, guild)
   122→            return ("❌ Batao kis user ko mute karna hai?", False, None)
   123→        
   124→        # ===== BOT CONTROL COMMANDS (Owner Only) =====
   125→        
   126→        if self._matches_patterns(msg_lower, ["status", "set status", "playing", "listening", "watching", "streaming"]):
   127→            if not self._check_owner(author):
   128→                return ("❌ Sirf owners status change kar sakte hain!", False, None)
   129→            activity_data = self._extract_activity(original_msg)
   130→            return await self._cmd_set_status(activity_data)
   131→        
   132→        if self._matches_patterns(msg_lower, ["nickname", "name change", "naam badlo", "call me", "bolte ho"]):
   133→            if not self._check_owner(author):
   134→                return ("❌ Sirf owners nickname change kar sakte hain!", False, None)
   135→            new_name = self._extract_nickname(original_msg)
   136→            if new_name:
   137→                return await self._cmd_set_nickname(new_name, guild)
   138→            return ("❌ Naya naam batao?", False, None)
   139→        
   140→        if self._matches_patterns(msg_lower, ["channel banao", "make channel", "new channel", "naya channel", "create channel"]):
   141→            if not self._check_owner(author):
   142→                return ("❌ Sirf channels bana sakte hain owners!", False, None)
   143→            channel_info = self._extract_channel_info(original_msg)
   144→            return await self._cmd_create_channel(channel_info, guild)
   145→        
   146→        if self._matches_patterns(msg_lower, ["role banao", "make role", "new role", "naya role", "create role"]):
   147→            if not self._check_owner(author):
   148→                return ("❌ Sirf roles bana sakte hain owners!", False, None)
   149→            role_info = self._extract_role_info(original_msg)
   150→            return await self._cmd_create_role(role_info, guild)
   151→        
   152→        # ===== MESSAGING COMMANDS =====
   153→        
   154→        if self._matches_patterns(msg_lower, ["bhejo", "send to", "message in", "channel me bhejo", "announce"]):
   155→            if not self._check_mod(author, guild) and "announce" in msg_lower:
   156→                pass  # Allow announces for mods
   157→            send_info = self._extract_send_info(original_msg, guild)
   158→            if send_info:
   159→                return await self._cmd_send_message(send_info, author)
   160→            return ("❌ Kya bhejna hai aur kahan? Specify karo!", False, None)
   161→        
   162→        if self._matches_patterns(msg_lower, ["embed", "rich embed", "fancy message", "beautiful message"]):
   163→            embed_info = self._extract_embed_info(original_msg)
   164→            if embed_info:
   165→                return await self._cmd_send_embed(embed_info, channel)

   171→            emoji = self._extract_emoji(original_msg)
   172→            target_msg = referenced_msg
   173→            if emoji and target_msg:
   174→                return await self._cmd_add_reaction(emoji, target_msg)
   175→            return ("❌ Emoji batao aur message pe reply karke bol!", False, None)
   176→        
   177→        # ===== UTILITY COMMANDS =====
   178→        
   179→        if self._matches_patterns(msg_lower, ["clear", "delete messages", "messages delete", "chat clean"]):
   180→            if not self._check_mod(author, guild):
   181→                return ("❌ Sirf mods messages delete kar sakte hain!", False, None)
   182→            count = self._extract_number(original_msg) or 10
   183→            return await self._cmd_clear_messages(count, channel)
   184→        
   185→        if self._matches_patterns(msg_lower, ["help", "commands", "kya kar sakti hai", "features"]):
   186→            return await self._cmd_show_help(author)

   188→        return None

   190→    def _matches_patterns(self, text: str, patterns: List[str]) -> bool:
   191→        """Check if text matches any of the given patterns"""
   192→        for pattern in patterns:
   193→            if pattern.lower() in text:
   194→                return True
   195→        return False

   197→    def _extract_target_user(
   198→        self, 
   199→        message: str, 
   200→        channel: discord.TextChannel, 
   201→        referenced_msg: Optional[discord.Message] = None
   202→    ) -> Optional[discord.Member]:
   203→        """Extract target user from mentions or referenced message"""
   204→        # First check for mentions in message
   205→        mention_pattern = r'<@!?(\d{17,19})>'
   206→        match = re.search(mention_pattern, message)
   207→        if match:
   208→            user_id = int(match.group(1))
   209→            return channel.guild.get_member(user_id)
   210→        
   211→        # Then check referenced message (reply)
   212→        if referenced_msg:
   213→            return channel.guild.get_member(referenced_msg.author.id)
   214→        
   215→        return None

   217→    def _extract_duration(self, message: str) -> int:
   218→        """Extract timeout duration from message in minutes"""
   219→        # Look for duration patterns
   220→        patterns = [
   221→            r'(\d+)\s*min',      # 10 min, 5min
   222→            r'(\d+)\s*minute',   # 10 minute
   223→            r'(\d+)\s*hour',     # 1 hour
   224→            r'(\d+)\s*hr',       # 1hr
   225→            r'(\d+)\s*day',      # 1 day
   226→        ]
   227→        
   228→        for pattern in patterns:
   229→            match = re.search(pattern, message.lower())
   230→            if match:
   231→                num = int(match.group(1))
   232→                if 'hour' in pattern or 'hr' in pattern:
   233→                    return num * 60
   234→                elif 'day' in pattern:
   235→                    return num * 1440
   236→                return num
   237→        
   238→        return 10  # Default 10 minutes

   240→    def _extract_reason(self, message: str) -> str:
   241→        """Extract reason from message"""
   242→        # Remove common prefixes to get reason
   243→        reason = message
   244→        for prefix in ['timeout', 'kick', 'ban', 'mute', 'do', 'kar', 'karo']:
   245→            reason = re.sub(prefix, '', reason, flags=re.IGNORECASE)
   246→        
   247→        # Remove mentions and numbers (duration)
   248→        reason = re.sub(r'<@!?\d+>', '', reason)
   249→        reason = re.sub(r'\d+\s*(min|hour|hr|day|minute)?', '', reason, flags=re.IGNORECASE)
   250→        
   251→        return reason.strip() or "No reason provided"

   253→    def _extract_activity(self, message: str) -> Dict[str, str]:
   254→        """Extract activity type and name from message"""
   255→        activity_type = "playing"  # Default
   256→        
   257→        if any(word in message.lower() for word in ['listening']):
   258→            activity_type = "listening"
   259→        elif any(word in message.lower() for word in ['watching']):
   260→            activity_type = "watching"
   261→        elif any(word in message.lower() for word in ['streaming']):
   262→            activity_type = "streaming"
   263→        
   264→        # Extract what they're doing
   265→        name_match = re.search(r'(?:playing|listening||watching|streaming|status)\s+(.+)', message, re.IGNORECASE)
   266→        name = name_match.group(1).strip() if name_match else "with Ophelia AI"
   267→        
   268→        return {"type": activity_type, "name": name}

   270→    def _extract_nickname(self, message: str) -> Optional[str]:
   271→        """Extract new nickname from message"""
   272→        # Look for quoted string or after keywords
   273→        quote_match = re.search(r'"([^"]+)"', message)
   274→        if quote_match:
   275→            return quote_match.group(1)
   276→        
   277→        # Try extracting after keywords
   278→        patterns = [
   279→            r'(?:nickname|name|naam|bolte)\s+(?:to|as|)?\s*(.+)',
   280→            r'(?:call me|bulao|bolo)\s*(.+)',
   281→        ]
   282→        for pattern in patterns:
   283→            match = re.search(pattern, message, re.IGNORECASE)
   284→            if match:
   285→                return match.group(1).strip()[:32]  # Discord max 32 chars
   286→        
   287→        return None

   289→    def _extract_channel_info(self, message: str) -> Dict[str, str]:
   290→        """Extract channel creation info"""
   291→        name_match = re.search(r'(?:channel)?\s*["\']?(\w[\w\s-]*)["\']?', message, re.IGNORECASE)
   292→        name = name_match.group(1).strip().lower().replace(" ", "-") if name_match else "general"
   293→        
   294→        ch_type = "text"
   295→        if 'voice' in message.lower():
   296→            ch_type = "voice"
   297→        elif 'stage' in message.lower():
   298→            ch_type = "stage"
   299→        
   300→        return {"name": name, "type": ch_type}
   301→    
   302→    def _extract_role_info(self, message: str) -> Dict[str, str]:
   303→        """Extract role creation info"""
   304→        name_match = re.search(r'(?:role)?\s*["\']?(\w[\w\s]*)["\']?', message, re.IGNORECASE)
   305→        name = name_match.group(1).strip() if name_match else "New Role"
   306→        
   307→        colors = {
   308→            'red': discord.Color.red(),
   309→            'blue': discord.Color.blue(),
   310→            'green': discord.Color.green(),
   311→            'yellow': discord.Color.gold(),
   312→            'purple': discord.Color.purple(),
   313→            'pink': discord.Color.pink(),
   314→            'orange': discord.Color.orange(),
   315→            'white': discord.Color.white(),
   316→            'black': discord.Color.default(),
   317→            'blurple': discord.Color.blurple(),
   318→            'gold': discord.Color.gold(),
   319→        }
   320→        
   321→        color = discord.Color.blue()  # Default
   322→        for color_name, color_val in colors.items():
   323→            if color_name in message.lower():
   324→                color = color_val
   325→                break
   326→        
   327→        return {"name": name, "color": color}

   329→    def _extract_send_info(self, message: str, guild: discord.Guild) -> Optional[Dict]:
   330→        """Extract message sending info"""
   331→        # Find channel
   332→        channel_match = re.search(r'<#(\d{17,19})>|(?:channel|#)(?:\s+me)?\s*(\w+)', message, re.IGNORECASE)
   333→        
   334→        target_channel = None
   335→        if channel_match:
   336→            if channel_match.group(1):  # Channel mention
   337→                target_channel = guild.get_channel(int(channel_match.group(1)))
   338→            else:  # Channel name
   339→                ch_name = channel_match.group(2).lower()
   340→                target_channel = discord.utils.get(guild.text_channels, name=ch_name)
   341→        
   342→        # Extract message content
   343→        content_match = re.search(r'["\'](.+)["\']|bhejo\s+(.+)|send\s+(.+)|announce\s+(.+)', message, re.IGNORECASE)
   344→        content = content_match.group(1) or content_match.group(2) or content_match.group(3) or content_match.group(4)
   345→        
   346→        if target_channel and content:
   347→            return {"channel": target_channel, "content": content.strip()}
   348→        
   349→        return None

   351→    def _extract_embed_info(self, message: str) -> Optional[Dict]:
   352→        """Extract embed creation info"""
   353→        # Look for title and description
   354→        matches = re.findall(r'"([^"]+)"', message)
   355→        if len(matches) >= 2:
   356→            return {"title": matches[0], "description": matches[1]}
   357→        elif len(matches) == 1:
   358→            return {"title": "Announcement", "description": matches[0]}
   359→        return None

   361→    def _extract_emoji(self, message: str) -> Optional[str]:
   362→        """Extract emoji from message"""
   363→        # Custom emoji format
   364→        custom_emoji = re.search(r'<a?:\w+:(\d+)>', message)
   365→        if custom_emoji:
   366→            return custom_emoji.group(0)
   367→        
   368→        # Common emojis - just return first emoji-like character found
   369→        emoji_chars = ['😂', '❤️', '🔥', '💀', '✨', '🎉', '👍', '👎', '😢', '😡', '🤔', '💯', '⭐', '🎮', '🎵']
   370→        for emoji in emoji_chars:
   371→            if emoji in message:
   372→                return emoji
   373→        
   374→        return None
   375→    
   376→    def _extract_number(self, message: str) -> Optional[int]:
   377→        """Extract number from message"""
   378→        match = re.search(r'\d+', message)
   379→        return int(match.group()) if match else None

   381→    def _check_owner(self, user: discord.Member) -> bool:
   382→        """Check if user is bot owner"""

   385→    def _check_mod(self, user: discord.Member, guild: discord.Guild) -> bool:
   386→        """Check if user is mod or owner"""
   387→        if is_owner(user.id):
   388→            return True
   389→        return user.guild_permissions.manage_guild or user.guild_permissions.administrator
