    54→    async def _detect_and_execute(

   177→    def _matches_patterns(self, text: str, patterns: List[str]) -> bool:
   178→        """Check if text matches any of the given patterns"""
   179→        for pattern in patterns:
   180→            if pattern.lower() in text:
   181→                return True
   182→        return False

   184→    async def _extract_target_user(
   185→        self, 
   186→        message: str, 
   187→        channel: discord.TextChannel, 
   188→        guild: discord.Guild,
   189→        referenced_msg: Optional[discord.Message] = None
   190→    ) -> Optional[discord.Member]:
   191→        """Extract target user from mentions, username, or referenced message - IMPROVED!"""
   192→        
   193→        # First check for mentions in message (most reliable)
   194→        mention_pattern = r'<@!?(\d{17,19})>'
   195→        match = re.search(mention_pattern, message)
   196→        if match:
   197→            user_id = int(match.group(1))
   198→            member = channel.guild.get_member(user_id)
   199→            if member:
   200→                return member
   201→        
   202→        # Then check referenced message (reply)
   203→        if referenced_msg:
   204→            return channel.guild.get_member(referenced_msg.author.id)
   205→        
   206→        # NEW: Try to find user by NAME/USERNAME from the message
   207→        # Remove common words and try to match a username
   208→        cleaned_msg = re.sub(r'<@!?\d+>', '', message)  # Remove mentions
   209→        cleaned_msg = re.sub(r'(kick|ban|timeout|mute|isko|iska|usko|use|inhe|inhone|user|ko|ka|ke|liye|karo|kar|do|to|me|se|hai|please|hey|ophelia|@everyone|@here)', '', cleaned_msg, flags=re.IGNORECASE)
   210→        cleaned_msg = cleaned_msg.strip()
   211→        
   212→        if cleaned_msg and len(cleaned_msg) > 2:
   213→            # Search for member by display name or username
   214→            for member in guild.members:
   215→                if cleaned_msg.lower() in member.display_name.lower() or cleaned_msg.lower() in member.name.lower():
   216→                    logger.info(f"🔍 Found user '{cleaned_msg}' → {member} ({member.id})")
   217→                    return member
   218→        
   219→        return None

   221→    def _extract_duration(self, message: str) -> int:
   222→        """Extract timeout duration from message in minutes"""
   223→        patterns = [
   224→            r'(\d+)\s*min',
   225→            r'(\d+)\s*minute',
   226→            r'(\d+)\s*hour',
   227→            r'(\d+)\s*hr',
   228→            r'(\d+)\s*day',
   229→        ]
   230→        
   231→        for pattern in patterns:
   232→            match = re.search(pattern, message.lower())
   233→            if match:
   234→                num = int(match.group(1))
   235→                if 'hour' in pattern or 'hr' in pattern:
   236→                    return num * 60
   237→                elif 'day' in pattern:
   238→                    return num * 1440
   239→                return num
   240→        
   241→        return 10  # Default 10 minutes

   243→    def _extract_reason(self, message: str) -> Optional[str]:
   244→        """Extract reason from message - returns None if no clear reason"""
   245→        reason = message
   246→        for prefix in ['timeout', 'kick', 'ban', 'mute', 'do', 'kar', 'karo']:
   247→            reason = re.sub(prefix, '', reason, flags=re.IGNORECASE)
   248→        
   249→        # Remove mentions and numbers (duration)
   250→        reason = re.sub(r'<@!?\d+>', '', reason)
   251→        reason = re.sub(r'\d+\s*(min|hour|hr|day|minute)?', '', reason, flags=re.IGNORECASE)
   252→        
   253→        reason = reason.strip()
   254→        
   255→        # Only return reason if it looks like actual content (not just random words)
   256→        if len(reason) < 3 or reason.lower() in ['isco', 'iska', 'usko', 'use', 'inhe', 'ko', 'se']:
   257→            return None  # No valid reason found
   258→        
   259→        return reason

   261→    def _extract_activity(self, message: str) -> Dict[str, str]:
   262→        """Extract activity type and name from message"""
   263→        activity_type = "playing"  # Default
   264→        
   265→        if any(word in message.lower() for word in ['listening']):
   266→            activity_type = "listening"
   267→        elif any(word in message.lower() for word in ['watching']):
   268→            activity_type = "watching"
   269→        elif any(word in message.lower() for word in ['streaming']):
   270→            activity_type = "streaming"
   271→        
   272→        name_match = re.search(r'(?:playing|listening||watching|streaming|status)\s+(.+)', message, re.IGNORECASE)
   273→        name = name_match.group(1).strip() if name_match else "with Ophelia AI"
   274→        
   275→        return {"type": activity_type, "name": name}

   277→    def _extract_nickname(self, message: str) -> Optional[str]:
   278→        """Extract new nickname from message"""
   279→        quote_match = re.search(r'"([^"]+)"', message)
   280→        if quote_match:
   281→            return quote_match.group(1)
   282→        
   283→        patterns = [
   284→            r'(?:nickname|name|naam|bolte)\s+(?:to|as|)?\s*(.+)',
   285→            r'(?:call me|bulao|bolo)\s*(.+)',
   286→        ]
   287→        for pattern in patterns:
   288→            match = re.search(pattern, message, re.IGNORECASE)
   289→            if match:
   290→                return match.group(1).strip()[:32]
   291→        
   292→        return None

   294→    def _extract_channel_info(self, message: str) -> Dict[str, str]:
   295→        """Extract channel creation info"""
   296→        name_match = re.search(r'(?:channel)?\s*["\']?(\w[\w\s-]*)["\']?', message, re.IGNORECASE)
   297→        name = name_match.group(1).strip().lower().replace(" ", "-") if name_match else "general"
   298→        
   299→        ch_type = "text"
   300→        if 'voice' in message.lower():
   301→            ch_type = "voice"
   302→        elif 'stage' in message.lower():
   303→            ch_type = "stage"
   304→        
   305→        return {"name": name, "type": ch_type}

   307→    def _extract_role_info(self, message: str) -> Dict[str, str]:
   308→        """Extract role creation info"""
   309→        name_match = re.search(r'(?:role)?\s*["\']?(\w[\w\s]*)["\']?', message, re.IGNORECASE)
   310→        name = name_match.group(1).strip() if name_match else "New Role"
   311→        
   312→        colors = {
   313→            'red': discord.Color.red(),
   314→            'blue': discord.Color.blue(),
   315→            'green': discord.Color.green(),
   316→            'yellow': discord.Color.gold(),
   317→            'purple': discord.Color.purple(),
   318→            'pink': discord.Color.pink(),
   319→            'orange': discord.Color.orange(),
   320→            'white': discord.Color.white(),
   321→            'black': discord.Color.default(),
   322→            'blurple': discord.Color.blurple(),
   323→            'gold': discord.Color.gold(),
   324→        }
   325→        
   326→        color = discord.Color.blue()
   327→        for color_name, color_val in colors.items():
   328→            if color_name in message.lower():
   329→                color = color_val
   330→                break
   331→        
   332→        return {"name": name, "color": color}

   334→    def _extract_send_info(self, message: str, guild: discord.Guild) -> Optional[Dict]:
   335→        """Extract message sending info"""
   336→        channel_match = re.search(r'<#(\d{17,19})>|(?:channel|#)(?:\s+me)?\s*(\w+)', message, re.IGNORECASE)
   337→        
   338→        target_channel = None
   339→        if channel_match:
   340→            if channel_match.group(1):
   341→                target_channel = guild.get_channel(int(channel_match.group(1)))
   342→            else:
   343→                ch_name = channel_match.group(2).lower()
   344→                target_channel = discord.utils.get(guild.text_channels, name=ch_name)
   345→        
   346→        content_match = re.search(r'["\'](.+)["\']|bhejo\s+(.+)|send\s+(.+)|announce\s+(.+)', message, re.IGNORECASE)
   347→        content = content_match.group(1) or content_match.group(2) or content_match.group(3) or content_match.group(4)
   348→        
   349→        if target_channel and content:
   350→            return {"channel": target_channel, "content": content.strip()}
   351→        
   352→        return None

   354→    def _extract_embed_info(self, message: str) -> Optional[Dict]:
   355→        """Extract embed creation info"""
   356→        matches = re.findall(r'"([^"]+)"', message)
   357→        if len(matches) >= 2:
   358→            return {"title": matches[0], "description": matches[1]}
   359→        elif len(matches) == 1:
   360→            return {"title": "Announcement", "description": matches[0]}
   361→        return None

   363→    def _extract_emoji(self, message: str) -> Optional[str]:
   364→        """Extract emoji from message"""
   365→        custom_emoji = re.search(r'<a?:\w+:(\d+)>', message)
   366→        if custom_emoji:
   367→            return custom_emoji.group(0)
   368→        
   369→        emoji_chars = ['😂', '❤️', '🔥', '💀', '✨', '🎉', '👍', '👎', '😢', '😡', '🤔', '💯', '⭐', '🎮', '🎵']
   370→        for emoji in emoji_chars:
   371→            if emoji in message:
   372→                return emoji
   373→        
   374→        return None

   376→    def _extract_number(self, message: str) -> Optional[int]:
   377→        """Extract number from message"""
   378→        match = re.search(r'\d+', message)
   379→        return int(match.group()) if match else None
