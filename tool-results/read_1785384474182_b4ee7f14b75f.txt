   662→   662→    def _ensure_payload_limit(self, messages: List[Dict], max_chars: int = None) -> List[Dict]:
   663→   663→        """Ensure total payload doesn't exceed Groq's limit"""
   664→   664→        if max_chars is None:
   665→   665→            max_chars = self.MAX_TOTAL_CHARS
   666→   666→        
   667→   667→        total_size = sum(len(m.get("content", "")) for m in messages)
   668→   668→        
   669→   669→        if total_size <= max_chars:
   670→   670→            return messages
   671→   671→        
   672→   672→        logger.warning(f"⚠️ Payload too large ({total_size} chars), trimming...")
   673→   673→        
   674→   674→        system_msgs = [m for m in messages if m.get("role") == "system"]
   675→   675→        other_msgs = [m for m in messages if m.get("role") != "system"]
   676→   676→        
   677→   677→        available_space = max_chars - sum(len(m.get("content", "")) for m in system_msgs)
   678→   678→        
   679→   679→        if available_space <= 0:
   680→   680→            logger.error("❌ System prompts too large, using minimal prompt")
   681→   681→            return [{"role": "system", "content": "You are Ophelia AI, a helpful Discord bot with personality."}]
   682→   682→        
   683→   683→        trimmed_msgs = []
   684→   684→        current_size = 0
   685→   685→        
   686→   686→        for msg in reversed(other_msgs):
   687→   687→            content = msg.get("content", "")
   688→   688→            if current_size + len(content) <= available_space:
   689→   689→                trimmed_msgs.insert(0, msg)
   690→   690→                current_size += len(content)
   691→   691→            else:
   692→   692→                break
   693→   693→        
   694→   694→        final_messages = system_msgs + trimmed_msgs
   695→   695→        logger.info(f"✅ Trimmed payload from {total_size} to {sum(len(m.get('content','')) for m in final_messages)} chars")
   696→   696→        
   697→   697→        return final_messages
