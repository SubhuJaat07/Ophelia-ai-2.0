   221→   221→    def _extract_duration(self, message: str) -> int:
   222→   222→        """Extract timeout duration from message in minutes"""
   223→   223→        patterns = [
   224→   224→            r'(\d+)\s*min',
   225→   225→            r'(\d+)\s*minute',
   226→   226→            r'(\d+)\s*hour',
   227→   227→            r'(\d+)\s*hr',
   228→   228→            r'(\d+)\s*day',
   229→   229→        ]
   230→   230→        
   231→   231→        for pattern in patterns:
   232→   232→            match = re.search(pattern, message.lower())
   233→   233→            if match:
   234→   234→                num = int(match.group(1))
   235→   235→                if 'hour' in pattern or 'hr' in pattern:
   236→   236→                    return num * 60
   237→   237→                elif 'day' in pattern:
   238→   238→                    return num * 1440
   239→   239→                return num
   240→   240→        
   241→   241→        return 10  # Default 10 minutes

   243→   243→    def _extract_reason(self, message: str) -> Optional[str]:
   244→   244→        """Extract reason from message - returns None if no clear reason"""
   245→   245→        reason = message
   246→   246→        for prefix in ['timeout', 'kick', 'ban', 'mute', 'do', 'kar', 'karo']:

   250→   250→        reason = re.sub(r'<@!?\d+>', '', reason)
   251→   251→        reason = re.sub(r'\d+\s*(min|hour|hr|day|minute)?', '', reason, flags=re.IGNORECASE)
   252→   252→        
   253→   253→        reason = reason.strip()
   254→   254→        
   255→   255→        # Only return reason if it looks like actual content (not just random words)
   256→   256→        if len(reason) < 3 or reason.lower() in ['isco', 'iska', 'usko', 'use', 'inhe', 'ko', 'se']:
   257→   257→            return None  # No valid reason found

   259→   259→        return reason
