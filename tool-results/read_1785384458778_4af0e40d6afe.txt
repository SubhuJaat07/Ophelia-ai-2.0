   130→    def _detect_mood(self, message: str) -> str:
   131→        """
   132→        Detect user's mood from message - EMOTIONAL INTELLIGENCE!
   133→        Returns: happy, sad, angry, excited, bored, confused, neutral, sarcastic
   134→        """
   135→        msg_lower = message.lower()
   136→        
   137→        # Happy indicators
   138→        happy_words = ["😂", "😁", "😄", "haha", "lol", "lmao", "nice", "great", "amazing", 
   139→                      "love", "best", "fire", "lit", "awesome", "happy", "excited", "yay",
   140→                      "❤️", "💕", "💖", "🔥", "✨", "🎉"]
   141→        if any(word in msg_lower for word in happy_words):
   142→            return "happy"
   143→        
   144→        # Sad indicators  
   145→        sad_words = ["😢", "😞", "😔", "sad", "depressed", "cry", "crying", "hurt",
   146→                    "pain", "alone", "lonely", "miss", "gone", "leave", "bye 😢"]
   147→        if any(word in msg_lower for word in sad_words):
   148→            return "sad"
   149→        
   150→        # Angry indicators
   151→        angry_words = ["😡", "😠", "angry", "mad", "hate", "stupid", "dumb", "idiot",
   152→                      "annoying", "frustrated", "wtf", "damn", "hell", "freaking"]
   153→        if any(word in msg_lower for word in angry_words):
   154→            return "angry"
   155→        
   156→        # Excited indicators
   157→        excited_words = ["!!!", "???", "omg", "oh my god", "wow", "can't believe",
   158→                        "finally", "let's go", "yesss", "yessss", "omggg"]
   159→        if any(word in msg_lower for word in excited_words):
   160→            return "excited"
   161→        
   162→        # Bored indicators
   163→        bored_words = ["bored", "boring", "nothing to do", "meh", "whatever", "ok",
   164→                      "fine", "k", "hmm", "sigh", "ugh"]
   165→        if any(word in msg_lower for word in bored_words):
   166→            return "bored"
   167→        
   168→        # Confused indicators
   169→        confused_words = ["?", "what", "how", "why", "don't understand", "confused",
   170→                        "explain", "help", "idk", "i don't know", "kaise", "kyun"]
   171→        if any(word in msg_lower for word in confused_words):
   172→            return "confused"
   173→        
   174→        # Sarcastic indicators (tricky!)
   175→        sarcastic_patterns = [r"oh really", r"wow.*so.*interesting", r"great.*job",
   176→                            r"thanks.*a lot", r"brilliant", r"sure.*jan", r"ha.*ha"]
   177→        if any(re.search(pattern, msg_lower) for pattern in sarcastic_patterns):
   178→            return "sarcastic"
   179→        
   180→        return "neutral"

   469→    def store_channel_message(
   470→        self,
   471→        channel_id: int,
   472→        author_name: str,
   473→        content: str,
   474→        is_bot: bool = False,
   475→        timestamp: str = None
   476→    ):
   477→        """
   478→        Store channel messages for CONTEXT AWARENESS!
   479→        This lets Ophelia know what's happening in the server!
   480→        """
   481→        self._init_clients()
   482→        
   483→        if not timestamp:
   484→            timestamp = datetime.now().isoformat()
   485→        
   486→        # Get existing channel history or create new
   487→        cache_key = f"channel_history:{channel_id}"
   488→        history = self.cache.get_user_context(channel_id)  # Reusing user_context for channel
   489→        
   490→        if not history:
   491→            history = {
   492→                "messages": [],
   493→                "participants": set(),
   494→                "topics": [],
   495→                "conflict_detected": False
   496→            }
   497→        
   498→        # Add new message
   499→        msg_entry = {
   500→            "author": author_name,
   501→            "content": content[:200],  # Truncate long messages
   502→            "is_bot": is_bot,
   503→            "timestamp": timestamp
   504→        }
   505→        
   506→        if isinstance(history, dict):
   507→            history["messages"].append(msg_entry)
   508→

   662→    def _ensure_payload_limit(self, messages: List[Dict], max_chars: int = None) -> List[Dict]:
   663→        """Ensure total payload doesn't exceed Groq's limit"""
   664→        if max_chars is None:
   665→            max_chars = self.MAX_TOTAL_CHARS
   666→        
   667→        total_size = sum(len(m.get("content", "")) for m in messages)
   668→        
   669→        if total_size <= max_chars:
   670→            return messages
   671→        
   672→        logger.warning(f"⚠️ Payload too large ({total_size} chars), trimming...")
   673→        
   674→        system_msgs = [m for m in messages if m.get("role") == "system"]
   675→        other_msgs = [m for m in messages if m.get("role") != "system"]
   676→        
   677→        available_space = max_chars - sum(len(m.get("content", "")) for m in system_msgs)
   678→        
   679→        if available_space <= 0:
   680→            logger.error("❌ System prompts too large, using minimal prompt")
   681→            return [{"role": "system", "content": "You are Ophelia AI, a helpful Discord bot with personality."}]
   682→        
   683→        trimmed_msgs = []
   684→        current_size = 0
   685→        
   686→        for msg in reversed(other_msgs):
   687→            content = msg.get("content", "")
   688→            if current_size + len(content) <= available_space:
   689→                trimmed_msgs.insert(0, msg)
   690→                current_size += len(content)
   691→            else:
   692→                break
   693→        
   694→        final_messages = system_msgs + trimmed_msgs
   695→        logger.info(f"✅ Trimmed payload from {total_size} to {sum(len(m.get('content','')) for m in final_messages)} chars")
   696→        
   697→        return final_messages
