    26→# 🔍 SEARCH MESSAGES TOOL
    27→# ============================================================
    28→class SearchMessagesTool(DiscordTool):
    29→    """
    30→    Search messages in server/channel - POWERFUL context finder!
    31→    
    32→    Lets AI find old conversations, specific topics, user messages.
    33→    """
    34→    
    35→    name = "search_messages"
    36→    description = """Search for messages in Discord server or channel. 
    37→Can find messages by content, author, date range, or keywords.
    38→Perfect for finding past discussions, context, or specific information.
    39→Supports filtering by: content (text search), author_id, channel_id, 
    40→limit (max results), and days_ago (how far back to search)."""
    41→    
    42→    parameters = [
    43→        ToolParameter(
    44→            name="query",
    45→            param_type="string",
    46→            description="Text/keywords to search for in messages",
    47→            required=True
    48→        ),
    49→        ToolParameter(
    50→            name="channel_id",
    51→            param_type="string", 
    52→            description="Specific channel ID to search (optional, searches all if omitted)",
    53→            required=False,
    54→            default=None
    55→        ),
    56→        ToolParameter(
    57→            name="author_id",
    58→            param_type="string",
    59→            description="Only search messages from this user ID",
    60→            required=False,
    61→            default=None
    62→        ),
    63→        ToolParameter(
    64→            name="limit",
    65→            param_type="integer",
    66→            description="Maximum number of results (default: 10, max: 50)",
    67→            required=False,
    68→            default=10
    69→        ),
    70→        ToolParameter(
    71→            name="days_ago",
    72→            param_type="integer",
    73→            description="How many days back to search (default: 30, max: 365)",
    74→            required=False,
    75→            default=30
    76→        )
    77→    ]
    78→    
    79→    permission_level = ToolPermissionLevel.EVERYONE
    80→    
    81→    async def execute(self, args: Dict[str, Any], context: Dict[str, Any]) -> ToolResult:
    82→        try:
    83→            query = args["query"].lower()
    84→            channel_id = args.get("channel_id")
    85→            author_id = args.get("author_id")
    86→            limit = min(args.get("limit", 10), 50)
    87→            days_ago = min(args.get("days_ago", 30), 365)
    88→            
    89→            guild = context.get("guild")
    90→            if not guild:
    91→                return ToolResult(success=False, content="❌ No guild/server available in context")
    92→            
    93→            # Calculate date threshold
    94→            since_date = datetime.utcnow() - timedelta(days=days_ago)
    95→            results = []
    96→            
    97→            # Determine channels to search
    98→            if channel_id:
    99→                channel = guild.get_channel(int(channel_id))
   100→                if not channel:
   101→                    return ToolResult(success=False, content=f"❌ Channel {channel_id} not found")
   102→                channels_to_search = [channel] if hasattr(channel, 'history') else []
   103→            else:
   104→                # Search all text channels bot can see
   105→                channels_to_search = [
   106→                    ch for ch in guild.text_channels 
   107→                    if ch.permissions_for(guild.me).read_message_history
   108→                ]
   109→            
   110→            # Search through channels
   111→            for channel in channels_to_search[:20]:  # Limit channels to search
   112→                try:
   113→                    async for message in channel.history(limit=100, after=since_date):
   114→                        # Apply filters
   115→                        if query and query not in message.content.lower():
   116→                            continue
   117→                        if author_id and str(message.author.id) != str(author_id):
   118→                            continue
   119→                        if message.author.bot:  # Skip bot messages
   120→                            continue
   121→                            
   122→                        results.append({
   123→                            "content": message.content[:300],
   124→                            "author": f"{message.author.display_name}",
   125→                            "author_id": str(message.author.id),
   126→                            "channel": f"#{message.channel.name}",
   127→                            "channel_id": str(message.channel.id),
   128→                            "message_id": str(message.id),
   129→                            "created_at": message.created_at.isoformat(),
   130→                            "jump_url": message.jump_url
   131→                        })
   132→                        
   133→                        if len(results) >= limit:
   134→                            break
   135→                    
   136→                    if len(results) >= limit:
   137→                        break
   138→                        
   139→                except Exception as e:
   140→                    logger.debug(f"Cannot search channel {channel}: {e}")
   141→                    continue
   142→            
   143→            if not results:
   144→                return ToolResult(
   145→                    success=True,
   146→                    content=f"😔 No messages found matching '{query}' in the last {days_ago} days.",
   147→                    data={"count": 0, "results": []}
   148→                )
   149→            
   150→            # Format results for AI
   151→            result_lines = [f"🔍 **Found {len(results)} message(s) about '{query}':**\n"]
   152→            for i, msg in enumerate(results[:10], 1):  # Show max 10 in summary
   153→                result_lines.append(
   154→                    f"{i}. **{msg['author']}** in {msg['channel']} ({msg['created_at'][:10]}):\n"
   155→                    f"   > {msg['content'][:200]}\n"
   156→                )
   157→            
   158→            return ToolResult(
   159→                success=True,
   160→                content="\n".join(result_lines),
   161→                data={"count": len(results), "results": results}
   162→            )
   163→            
   164→        except Exception as e:
   165→            logger.error(f"Search failed: {e}")
   166→            return ToolResult(success=False, content=f"❌ Search error: {str(e)[:150]}")

   170→# 📖 READ CHANNEL TOOL  
   171→# ============================================================
   172→class ReadChannelTool(DiscordTool):
   173→    """
   174→    Read recent messages from a channel - Context gathering!
   175→    
   176→    AI can use this to understand what's happening in any channel.
   177→    """
   178→    
   179→    name = "read_channel_messages"
   180→    description = """Read recent messages from a Discord channel.
   181→Useful for understanding current discussions, getting context, or catching up on conversations.
   182→Returns messages with author, timestamp, and content."""
   183→    
   184→    parameters = [
   185→        ToolParameter(
   186→            name="channel_id",
   187→            param_type="string",
   188→            description="ID of the channel to read from",
   189→            required=True
   190→        ),
   191→        ToolParameter(
   192→            name="limit",
   193→            param_type="integer",
   194→            description="Number of messages to fetch (default: 20, max: 100)",
   195→            required=False,
   196→            default=20
   197→        ),
   198→        ToolParameter(
   199→            name="before_message_id",
   200→            param_type="string",
   201→            description="Get messages before this message ID (for pagination)",
   202→            required=False,
   203→            default=None
   204→        )
   205→    ]
   206→    
   207→    permission_level = ToolPermissionLevel.EVERYONE
   208→    
   209→    async def execute(self, args: Dict[str, Any], context: Dict[str, Any]) -> ToolResult:
   210→        try:
   211→            channel_id = args["channel_id"]
   212→            limit = min(args.get("limit", 20), 100)
   213→            before_id = args.get("before_message_id")
   214→            
   215→            guild = context.get("guild")
   216→            if not guild:
   217→                return ToolResult(success=False, content="❌ No guild available")
   218→            
   219→            channel = guild.get_channel(int(channel_id))
   220→            if not channel:
   221→                return ToolResult(success=False, content=f"❌ Channel {channel_id} not found or not accessible")
   222→            
   223→            if not hasattr(channel, 'history'):
   224→                return ToolResult(success=False, content=f"❌ Cannot read from this channel type")
   225→            
   226→            # Check permissions
   227→            if not channel.permissions_for(guild.me).read_message_history:
   228→                return ToolResult(success=False, content="❌ No permission to read this channel's history")
   229→            
   230→            # Fetch messages
   231→            kwargs = {"limit": limit}
   232→            if before_id:
   233→                kwargs["before"] = discord.Object(id=int(before_id))
   234→            
   235→            messages = []
   236→            async for msg in channel.history(**kwargs):
   237→                messages.append(msg)
   238→            
   239→            if not messages:
   240→                return ToolResult(success=True, content=f"📭 No messages found in #{channel.name}", data=[])
   241→            
   242→            messages.reverse()  # Chronological order
   243→            
   244→            # Format for AI
   245→            lines = [f"📖 **Recent messages from #{channel.name}** ({len(messages)} messages):\n"]
   246→            for msg in messages[-30:]:  # Max 30 in output
   247→                is_bot = "🤖 " if msg.author.bot else ""
   248→                lines.append(
   249→                    f"[{msg.created_at.strftime('%H:%M')}] {is_bot}**{msg.author.display_name}:** {msg.content[:250]}"
   250→                )
   251→            
   252→            return ToolResult(
   253→                success=True,
   254→                content="\n".join(lines),
   255→                data={
   256→                    "count": len(messages),
   257→                    "channel_name": channel.name,
   258→                    "messages": [{
   259→                        "author": m.author.display_name,
   260→                        "content": m.content,
   261→                        "timestamp": m.created_at.isoformat()
   262→                    } for m in messages]
   263→                }
   264→            )
   265→            
   266→        except Exception as e:
   267→            logger.error(f"Read channel failed: {e}")
   268→            return ToolResult(success=False, content=f"❌ Error reading channel: {str(e)[:150]}")

   272→# 💬 SEND MESSAGE TOOL
   273→# ============================================================
   274→class SendMessageTool(DiscordTool):
   275→    """
   276→    Send a message to any channel - Proactive messaging!
   277→    
   278→    AI can use this to send announcements, replies, etc.
   279→    """
   280→    
   281→    name = "send_message"
   282→    description = """Send a message to a Discord channel.
   283→Use for sending announcements, responding in other channels, or proactive messaging.
   284→Can optionally reply to an existing message."""
   285→    
   286→    parameters = [
   287→        ToolParameter(
   288→            name="channel_id",
   289→            param_type="string",
   290→            description="ID of the channel to send message to",
   291→            required=True
   292→        ),
   293→        ToolParameter(
   294→            name="message",
   295→            param_type="string",
   296→            description="Content of the message to send",
   297→            required=True
   298→        ),
   299→        ToolParameter(
   300→            name="reply_to_message_id",
   301→            param_type="string",
   302→            description="Optional: Message ID to reply to",
   303→            required=False,
   304→            default=None
   305→        )
   306→    ]
   307→    
   308→    permission_level = ToolPermissionLevel.MODERATOR  # Require mod for safety
   309→    
   310→    async def execute(self, args: Dict[str, Any], context: Dict[str, Any]) -> ToolResult:
   311→        try:
   312→            channel_id = args["channel_id"]
   313→            content = args["message"][:1900]  # Discord limit
   314→            reply_to_id = args.get("reply_to_message_id")
   315→            
   316→            guild = context.get("guild")
   317→            if not guild:
   318→                return ToolResult(success=False, content="❌ No guild available")
   319→            
   320→            channel = guild.get_channel(int(channel_id))
   321→            if not channel:
   322→                return ToolResult(success=False, content=f"❌ Channel {channel_id} not found")
   323→            
   324→            # Check permissions
   325→            if not channel.permissions_for(guild.me).send_messages:
   326→                return ToolResult(success=False, content="❌ No permission to send messages here")
   327→            
   328→            # Send message
   329→            kwargs = {}
   330→            if reply_to_id:
   331→                try:
   332→                    reply_msg = await channel.fetch_message(int(reply_to_id))
   333→                    kwargs["reference"] = reply_msg.to_reference()
   334→                except:
   335→                    pass  # Continue without reply if message not found
   336→            
   337→            sent_msg = await channel.send(content=content, **kwargs)
   338→            
   339→            return ToolResult(
   340→                success=True,
   341→                content=f"✅ Message sent to #{channel.name}! (ID: {sent_msg.id})",
   342→                data={"message_id": str(sent_msg.id), "channel": channel.name}
   343→            )
   344→            
   345→        except discord.Forbidden:
   346→            return ToolResult(success=False, content="❌ Forbidden: Missing permissions to send message")
   347→        except Exception as e:
   348→            logger.error(f"Send message failed: {e}")
   349→            return ToolResult(success=False, content=f"❌ Error sending: {str(e)[:150]}")

   429→# ============================================================
   430→class GetMemberTool(DiscordTool):
   431→    """
   432→    Get detailed info about a server member - User lookup!
   433→    
   434→    AI can look up users by ID or mention.
   435→    """
   436→    
   437→    name = "get_member_info"
   438→    description = """Get detailed information about a Discord server member.
   439→Returns username, display name, roles, join date, status, and more.
   440→Use for looking up users, verifying members, or getting user context."""
   441→    
   442→    parameters = [
   443→        ToolParameter(
   444→            name="user_id",
   445→            param_type="string",
   446→            description="The user's Discord ID to look up",
   447→            required=True
   448→        ),
   449→        ToolParameter(
   450→            name="guild_id",
   451→            param_type="string",
   452→            description="Guild ID (uses current guild if omitted)",
   453→            required=False,
   454→            default=None
   455→        )
   456→    ]
   457→    
   458→    permission_level = ToolPermissionLevel.EVERYONE
   459→    
   460→    async def execute(self, args: Dict[str, Any], context: Dict[str, Any]) -> ToolResult:
   461→        try:
   462→            user_id = args["user_id"]
   463→            
   464→            guild = context.get("guild")
   465→            if not guild:
   466→                return ToolResult(success=False, content="❌ No guild available")
   467→            
   468→            # Try to get member
   469→            member = None
   470→            try:
   471→                member = await guild.fetch_member(int(user_id))
   472→            except discord.NotFound:
   473→                # Try get_member (cache only)
   474→                member = guild.get_member(int(user_id))
   475→            
   476→            if not member:
   477→                return ToolResult(success=False, content=f"❌ User {user_id} not found in this server")
   478→            
   479→            # Gather member info
   480→            roles = [role.name for role in member.roles if role.name != "@everyone"]
   481→            joined_at = member.joined_at.strftime("%Y-%m-%d") if member.joined_at else "Unknown"
   482→            created_at = member.created_at.strftime("%Y-%m-%d %H:%M") if member.created_at else "Unknown"
   483→            
   484→            # Status info
   485→            status = "🟢 Online" if member.status == discord.Status.online else \
   486→                     "🌙 Idle" if member.status == discord.Status.idle else \
   487→                     "🔴 DND" if member.status == discord.Status.dnd else \
   488→                     "⚫ Offline"
   489→            
   490→            # Activity info
   491→            activity_str = ""
   492→            if member.activity:
   493→                activity_str = f"\n🎮 Activity: {member.activity.name}" if member.activity.name else ""
   494→            
   495→            info = f"""👤 **Member Info:** {member.display_name}
   496→
   497→**Basic Info:**
   498→• Username: `{member.name}`
   499→• ID: `{member.id}`
   500→• Bot: {'✅ Yes' if member.bot else '❌ No'}
   501→• Account Created: {created_at}
   502→
   503→**Server Info:**
   504→• Nickname: {member.nick or 'None'}
   505→• Joined Server: {joined_at}
   506→• Status: {status}{activity_str}
   507→• Roles ({len(roles)}): {', '.join(roles[:10]) if roles else 'None'}
   508→
   509→**Avatar URL:** {member.avatar.url if member.avatar else 'Default'}"""
   510→            
   511→            return ToolResult(
   512→                success=True,
   513→                content=info,
   514→                data={
   515→                    "username": member.name,
   516→                    "display_name": member.display_name,
   517→                    "id": str(member.id),
   518→                    "roles": roles,
   519→                    "joined_at": joined_at,
   520→                    "is_bot": member.bot,
   521→                    "status": str(member.status)
   522→                }
   523→            )
   524→            
   525→        except Exception as e:
   526→            logger.error(f"Get member failed: {e}")
   527→            return ToolResult(success=False, content=f"❌ Error: {str(e)[:150]}")

   542→Returns server name, member count, channel count, features, creation date, and more.
   543→Helps AI understand the server context."""

   545→    parameters = [
   546→        ToolParameter(
   547→            name="guild_id",
   548→            param_type="string",
   549→            description="Guild ID (uses current guild if omitted)",
   550→            required=False,
   551→            default=None
   552→        )

   574→

   576→            features = ", ".join(guild.features[:8]) if guild.features else "None"
   577→            
   578→            info = f"""🏠 **Server Information**
   579→
   580→**Name:** {guild.name}
   581→**ID:** `{guild.id}`
   582→**Owner:** {guild.owner.display_name if guild.owner else 'Unknown'} (`{guild.owner_id}`)
   583→
   584→**Statistics:**
   585→• Members: {guild.member_count} total ({humans} humans, {bots} bots)
   586→• Online Now: ~{online}
   587→• Channels: {text_channels} text + {voice_channels} voice (in {categories} categories)
   588→• Roles: {roles}
   589→• Emojis: {emojis}
   590→• Stickers: {len(guild.stickers)}

   592→**Details:**

   595→• Content Filter: {str(guild.explicit_content_filter)}
   596→• Features: {features}
   597→
   598→**Bot Permissions Here:**
   599→{self._format_permissions(guild.me.guild_permissions)}"""
   600→            
   601→            return ToolResult(
   602→                success=True,
   603→                content=info,
   604→                data={
   605→                    "name": guild.name,
   606→                    "id": str(guild.id),
