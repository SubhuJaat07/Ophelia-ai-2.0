    19→def check_is_mod():
    20→    """Check if user has moderation permissions OR is owner"""
    21→    async def predicate(interaction: discord.Interaction) -> bool:
    22→        if not interaction.guild:
    23→            await interaction.response.send_message(
    24→                "❌ Ye command sirf servers me kaam karegi!", ephemeral=True
    25→            )
    26→            return False
    27→        
    28→        user = interaction.user
    29→        
    30→        # Owners always have access!
    31→        if is_owner(user.id):
    32→            return True
    33→        
    34→        # Check mod permissions
    35→        is_admin = interaction.guild.owner_id == user.id
    36→        has_mod_perms = user.guild_permissions.manage_guild or user.guild_permissions.administrator
    37→        
    38→        if not (is_admin or has_mod_perms):
    39→            await interaction.response.send_message(
    40→                "❌ Sirf Mods/Admins/Owners hi settings change kar sakte hain!", ephemeral=True
    41→            )
    42→            return False
    43→        
    44→        return True
    45→    
    46→    return app_commands.check(predicate)

   333→        max_length=4,

   338→        try:

   340→            temp = max(0.0, min(2.0, temp))  # Clamp between 0-2
   341→            
   342→            ai = get_ai_handler()
   343→            db = get_db()
   344→            cache = get_cache()
   345→            
   346→            settings = await ai.get_guild_settings(self.guild_id)
   347→            settings["temperature"] = temp
   348→            
   349→            cache.set_guild_settings(self.guild_id, settings)
   350→            await db.upsert_guild_settings(self.guild_id, settings)
   351→            
   352→            embed = self.parent_view._create_settings_embed(settings)
   353→            embed.add_field(
   354→                name="✅ Temperature Updated!",
   355→                value=f"Ab temperature **{temp}** hai!",
   356→                inline=False
   357→            )
   358→            await interaction.response.edit_message(embed=embed, view=self.parent_view)
