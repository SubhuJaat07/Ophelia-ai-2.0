   422→   422→    def _clean_ai_response(self, response: str) -> str:
   423→   423→        """
   424→   424→        Clean AI response - remove raw function calls and weird formatting.
   425→   425→        Prevents leaks like <function=get_channel_info[...]>
   426→   426→        """
   427→   427→        if not response:
   428→   428→            return response
   429→   429→        
   430→   430→        # Remove raw function call syntax: <function=name[params]>
   431→   431→        response = re.sub(r'<function=\w+[\[\{].*?[\]\}]>', '', response, flags=re.DOTALL)
   432→   432→        
   433→   433→        # Remove standalone function calls: function_name(params)
   434→   434→        response = re.sub(r'^\s*\w+\([^)]*\)\s*$', '', response, flags=re.MULTILINE)
   435→   435→
