    31→class SearchMessagesTool(DiscordTool):
    32→    """Search messages in server/channel"""
    33→    
    34→    name = "search_messages"
    35→    description = """Search for messages in Discord server or channel by content, author, or date."""
    36→    
    37→    parameters = [
    38→        ToolParameter(name="query", param_type="string", description="Text to search for", required=True),
    39→        ToolParameter(name="channel_id", param_type="string", description="Specific channel ID", required=False, default=None),
    40→        ToolParameter(name="author_id", param_type="string", description="Author user ID", required=False, default=None),
    41→        ToolParameter(name="limit", param_type="integer", description="Max results (default: 10)", required=False, default=10),
    42→    ]
    43→    
    44→    permission_level = ToolPermissionLevel.EVERYONE
    45→    
    46→    async def execute(self, args: Dict[str, Any], context: Dict[str, Any]) -> ToolResult:
    47→        try:
    48→            query = args["query"].lower()
    49→            channel_id = args.get("channel_id")
    50→            author_id = args.get("author_id")
    51→            limit = min(args.get("limit", 10), 50)
    52→            
    53→            guild = context.get("guild")
    54→            if not guild:
    55→                return ToolResult(success=False, content="❌ No guild available")
    56→            
    57→            since_date = datetime.utcnow() - timedelta(days=30)
    58→            results = []
    59→            
    60→            channels_to_search = []
    61→            if channel_id:
    62→                ch = guild.get_channel(int(channel_id))
    63→                channels_to_search = [ch] if ch and hasattr(ch, 'history') else []
    64→            else:
    65→                channels_to_search = [ch for ch in guild.text_channels if ch.permissions_for(guild.me).read_message_history]
    66→            
    67→            for channel in channels_to_search[:20]:
    68→                try:
    69→                    async for message in channel.history(limit=100, after=since_date):
    70→                        if query and query not in message.content.lower():
    71→                            continue
    72→                        if author_id and str(message.author.id) != str(author_id):
    73→                            continue
    74→                        if message.author.bot:
    75→                            continue
    76→                        results.append({
    77→                            "content": message.content[:300],
    78→                            "author": f"{message.author.display_name}",
    79→                            "channel": f"#{message.channel.name}",
    80→                            "message_id": str(message.id),
    81→                            "jump_url": message.jump_url
    82→                        })
    83→                        if len(results) >= limit:
    84→                            break
    85→                    if len(results) >= limit:
    86→                        break
    87→                except Exception as e:
    88→                    logger.debug(f"Cannot search channel {channel}: {e}")
    89→            
    90→            if not results:
    91→                return ToolResult(success=True, content=f"😔 No messages found matching '{query}'", data={"count": 0})
    92→            
    93→            lines = [f"🔍 **Found {len(results)} message(s):**\n"]
    94→            for i, msg in enumerate(results[:10], 1):
    95→                lines.append(f"{i}. **{msg['author']}** in {msg['channel']}: {msg['content'][:200]}")
    96→            
    97→            return ToolResult(success=True, content="\n".join(lines), data={"count": len(results), "results": results})
    98→        except Exception as e:
    99→            return ToolResult(success=False, content=f"❌ Search error: {str(e)[:150]}")

   102→class ReadChannelTool(DiscordTool):
   103→    """Read recent messages from a channel"""
   104→    
   105→    name = "read_channel_messages"
   106→    description = """Read recent messages from a Discord channel for context."""
   107→    
   108→    parameters = [
   109→        ToolParameter(name="channel_id", param_type="string", description="Channel ID to read", required=True),
   110→        ToolParameter(name="limit", param_type="integer", description="Number of messages (default: 20)", required=False, default=20),
   111→    ]
   112→    
   113→    permission_level = ToolPermissionLevel.EVERYONE
   114→    
   115→    async def execute(self, args: Dict[str, Any], context: Dict[str, Any]) -> ToolResult:
   116→        try:
   117→            channel_id = args["channel_id"]
   118→            limit = min(args.get("limit", 20), 100)
   119→            
   120→            guild = context.get("guild")
   121→            if not guild:
   122→                return ToolResult(success=False, content="❌ No guild available")
   123→            
   124→            channel = guild.get_channel(int(channel_id))
   125→            if not channel or not hasattr(channel, 'history'):
   126→                return ToolResult(success=False, content=f"❌ Channel not found or cannot read")
   127→            
   128→            if not channel.permissions_for(guild.me).read_message_history:
   129→                return ToolResult(success=False, content="❌ No permission to read history")
   130→            
   131→            messages = []
   132→            async for msg in channel.history(limit=limit):
   133→                messages.append(msg)
   134→            
   135→            if not messages:
   136→                return ToolResult(success=True, content=f"📭 No messages in #{channel.name}")
   137→            
   138→            messages.reverse()
   139→            lines = [f"📖 **#{channel.name}** ({len(messages)} messages):\n"]
   140→            for msg in messages[-30:]:
   141→                is_bot = "🤖 " if msg.author.bot else ""
   142→                lines.append(f"[{msg.created_at.strftime('%H:%M')}] {is_bot}**{msg.author.display_name}:** {msg.content[:250]}")
   143→            
   144→            return ToolResult(success=True, content="\n".join(lines), data={"count": len(messages)})
   145→        except Exception as e:
   146→            return ToolResult(success=False, content=f"❌ Error: {str(e)[:150]}")

   337→class SendEmbedTool(DiscordTool):
   338→    """Send a rich embed message"""
   339→    
   340→    name = "send_embed"
   341→    description = """Send a beautiful embedded message with title, description, colors, fields, images, etc."""
   342→    
   343→    parameters = [
   344→        ToolParameter(name="channel_id", param_type="string", description="Channel ID to send to", required=True),
   345→        ToolParameter(name="title", param_type="string", description="Embed title", required=True),
   346→        ToolParameter(name="description", param_type="string", description="Embed description/body", required=False, default=None),
   347→        ToolParameter(name="color", param_type="string", description="Color (hex like 'FF0000' or name like 'red', 'blue', 'green')", required=False, default="0x5865F2"),
   348→        ToolParameter(name="fields", param_type="string", description="Fields as JSON array: [{\"name\":\"Field\",\"value\":\"Value\"}]", required=False, default="[]"),
   349→        ToolParameter(name="footer", param_type="string", description="Footer text", required=False, default=None),
   350→        ToolParameter(name="thumbnail_url", param_type="string", description="Thumbnail image URL", required=False, default=None),
   351→        ToolParameter(name="image_url", param_type="string", description="Main image URL", required=False, default=None),
   352→        ToolParameter(name="author_name", param_type="string", description="Author name", required=False, default=None),
   353→    ]
   354→    
   355→    permission_level = ToolPermissionLevel.MODERATOR
   356→    
   357→    async def execute(self, args: Dict[str, Any], context: Dict[str, Any]) -> ToolResult:
   358→        try:
   359→            import json
   360→            
   361→            channel_id = args["channel_id"]
   362→            title = args["title"][:256]
   363→            description = args.get("description")
   364→            color_str = args.get("color", "0x5865F2")
   365→            fields_json = args.get("fields", "[]")
   366→            footer = args.get("footer")
   367→            thumbnail_url = args.get("thumbnail_url")
   368→            image_url = args.get("image_url")
   369→            author_name = args.get("author_name")
   370→            
   371→            # Parse color
   372→            color_map = {
   373→                "red": 0xFF0000, "green": 0x00FF00, "blue": 0x0000FF,
   374→                "yellow": 0xFFFF00, "purple": 0x9900FF, "orange": 0xFF8800,
   375→                "white": 0xFFFFFF, "black": 0x000000
   376→            }
   377→            if color_str.lower() in color_map:
   378→                color = color_map[color_str.lower()]
   379→            elif color_str.startswith("0x"):
   380→                color = int(color_str, 16)
   381→            elif color_str.startswith("#"):
   382→                color = int(color_str[1:], 16)
   383→            else:
   384→                color = 0x5865F2  # Discord blurple
   385→            
   386→            # Build embed
   387→            embed = discord.Embed(
   388→                title=title,
   389→                description=description,
   390→                color=color,
   391→                timestamp=datetime.utcnow()
   392→            )
   393→            
   394→            # Parse and add fields
   395→            try:
   396→                fields = json.loads(fields_json)
   397→                for field in fields[:25]:  # Max 25 fields
   398→                    inline = field.get("inline", False)
   399→                    embed.add_field(name=field["name"][:256], value=field["value"][:1024], inline=inline)
   400→            except json.JSONDecodeError:
   401→                pass
   402→            
   403→            if footer:
   404→                embed.set_footer(text=footer[:2048])
   405→            if thumbnail_url:
   406→                embed.set_thumbnail(url=thumbnail_url)
   407→            if image_url:
   408→                embed.set_image(url=image_url)
   409→            if author_name:
   410→                embed.set_author(name=author_name[:256])
   411→            
   412→            guild = context.get("guild")
   413→            if not guild:
   414→                return ToolResult(success=False, content="❌ No guild available")
   415→            
   416→            channel = guild.get_channel(int(channel_id))
   417→            if not channel:
   418→                return ToolResult(success=False, content="❌ Channel not found")
   419→            
   420→            msg = await channel(embed=embed)
   421→            return ToolResult(success=True, content=f"✅ Embed sent to #{channel.name}!", data={"message_id": str(msg.id)})
   422→        except Exception as e:
   423→            return ToolResult(success=False, content=f"❌ Embed failed: {str(e)[:200]}")
